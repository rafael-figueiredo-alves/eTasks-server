CREATE SCHEMA IF NOT EXISTS `etasks_server`
DEFAULT CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE `etasks_server`;

CREATE TABLE IF NOT EXISTS `version` (
    `Id` INT NOT NULL DEFAULT 1,
    `AppVersion` INT NOT NULL,
    `DisplayVersion` VARCHAR(20) NOT NULL,
    `URL_APK` VARCHAR(255) NOT NULL,
    `URL_Win` VARCHAR(255) NOT NULL,
    PRIMARY KEY (`Id`),
    UNIQUE INDEX `UX_version_id` (`Id` ASC)
) ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `users` (
    `Uid` CHAR(36) NOT NULL,

    `LastAccessAt` DATETIME NULL,

CREATE TABLE IF NOT EXISTS `task_items` (
    `Id` CHAR(36) NOT NULL,
    `UserUid` CHAR(36) NOT NULL,
    `GeneratedFromTaskId` CHAR(36) NULL,
    `Summary` VARCHAR(200) NOT NULL,
    `Notes` TEXT NULL,
    `Priority` INT NOT NULL DEFAULT 1,
    `TaskDate` DATETIME NOT NULL,
    `IsCompleted` TINYINT(1) NOT NULL DEFAULT 0,
    `CompletedAt` DATETIME NULL,
    `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `UpdatedAt` DATETIME NULL,
    `IsDeleted` TINYINT(1) NOT NULL DEFAULT 0,
    `DeletedAt` DATETIME NULL,
    PRIMARY KEY (`Id`),
    INDEX `IX_task_items_useruid_taskdate` (`UserUid` ASC, `TaskDate` ASC),
    INDEX `IX_task_items_useruid_iscompleted` (`UserUid` ASC, `IsCompleted` ASC),
    INDEX `IX_task_items_useruid_isdeleted` (`UserUid` ASC, `IsDeleted` ASC),
    INDEX `IX_task_items_generatedfromtaskid` (`GeneratedFromTaskId` ASC),
    CONSTRAINT `fk_task_items_users`
        FOREIGN KEY (`UserUid`) REFERENCES `users` (`Uid`)
