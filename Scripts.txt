-- MÓDULO DE AUTENTICAÇÃO
CREATE TABLE IF NOT EXISTS `users` (
  `Uid` CHAR(36) NOT NULL COMMENT 'UUID v7 string',
  `Name` VARCHAR(100) NOT NULL,
  `Email` VARCHAR(150) NOT NULL,
  `PasswordHash` VARCHAR(255) NOT NULL,
  `PhotoPath` VARCHAR(500) NULL,
  `IsConfirmed` TINYINT(1) NOT NULL DEFAULT 0,
  `IsAdmin` TINYINT(1) NOT NULL DEFAULT 0,
  `IsBlocked` TINYINT(1) DEFAULT 0,
  `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`Uid`),
  UNIQUE INDEX `Email_UNIQUE` (`Email` ASC)
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `refresh_tokens` (
  `Id` CHAR(36) NOT NULL COMMENT 'UUID v7 string',
  `UserUid` CHAR(36) NOT NULL,
  `Token` VARCHAR(255) NOT NULL,
  `UserAgent` VARCHAR(255) NULL COMMENT 'Identifica se é Delphi App ou Web Site',
  `ExpiresAt` DATETIME NOT NULL,
  `IsRevoked` TINYINT(1) NOT NULL DEFAULT 0,
  `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`Id`),
  CONSTRAINT `fk_refresh_tokens_users` FOREIGN KEY (`UserUid`) REFERENCES `users` (`Uid`) ON DELETE CASCADE
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `password_reset_codes` (
  `Id` CHAR(36) NOT NULL COMMENT 'UUID v7 string',
  `UserUid` CHAR(36) NOT NULL,
  `Code` VARCHAR(6) NOT NULL,
  `ExpiresAt` DATETIME NOT NULL,
  `IsUsed` TINYINT(1) NOT NULL DEFAULT 0,
  `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`Id`),
  CONSTRAINT `fk_password_reset_users` FOREIGN KEY (`UserUid`) REFERENCES `users` (`Uid`) ON DELETE CASCADE
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS login_logs (
    Id CHAR(36) NOT NULL,
    UserUid CHAR(36),
    Status VARCHAR(20) NOT NULL,
    IpAddress VARCHAR(50),
    UserAgent VARCHAR(255),
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (Id),
    CONSTRAINT fk_login_logs_user FOREIGN KEY (UserUid) REFERENCES users(Uid) ON DELETE SET NULL
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;
