-- MÓDULO DE AUTENTICAÇÃO
CREATE TABLE IF NOT EXISTS `users` (
  `Uid` CHAR(36) NOT NULL COMMENT 'UUID v7 string',
  `Name` VARCHAR(100) NOT NULL,
  `Email` VARCHAR(150) NOT NULL,
  `PasswordHash` VARCHAR(255) NOT NULL,
  `PhotoPath` VARCHAR(500) NULL,
  `IsConfirmed` TINYINT(1) NOT NULL DEFAULT 0,
  `IsAdmin` TINYINT(1) NOT NULL DEFAULT 0,
  `IsBlocked` TINYINT(1) DEFAULT 0,
  `LastAccessAt` DATETIME NULL,
  `IsDeleted` TINYINT(1) NOT NULL DEFAULT 0,
  `DeletedAt` DATETIME NULL,
  `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`Uid`),
  UNIQUE INDEX `Email_UNIQUE` (`Email` ASC)
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `refresh_tokens` (
  `Id` CHAR(36) NOT NULL COMMENT 'UUID v7 string',
  `UserUid` CHAR(36) NOT NULL,
  `Token` VARCHAR(255) NOT NULL,
  `UserAgent` VARCHAR(255) NULL COMMENT 'Identifica se é Delphi App ou Web Site',
  `ExpiresAt` DATETIME NOT NULL,
  `IsRevoked` TINYINT(1) NOT NULL DEFAULT 0,
  `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`Id`),
  CONSTRAINT `fk_refresh_tokens_users` FOREIGN KEY (`UserUid`) REFERENCES `users` (`Uid`) ON DELETE CASCADE
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `password_reset_codes` (
  `Id` CHAR(36) NOT NULL COMMENT 'UUID v7 string',
  `UserUid` CHAR(36) NOT NULL,
  `Code` VARCHAR(6) NOT NULL,
  `ExpiresAt` DATETIME NOT NULL,
  `IsUsed` TINYINT(1) NOT NULL DEFAULT 0,
  `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`Id`),
  CONSTRAINT `fk_password_reset_users` FOREIGN KEY (`UserUid`) REFERENCES `users` (`Uid`) ON DELETE CASCADE
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS login_logs (
    Id CHAR(36) NOT NULL,
    UserUid CHAR(36),
    Status VARCHAR(20) NOT NULL,
    IpAddress VARCHAR(50),
    UserAgent VARCHAR(255),
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (Id),
    CONSTRAINT fk_login_logs_user FOREIGN KEY (UserUid) REFERENCES users(Uid) ON DELETE SET NULL
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `user_settings` (
    `Id` CHAR(36) NOT NULL,
    `UserUid` CHAR(36) NOT NULL,
    `Theme` VARCHAR(10) NOT NULL DEFAULT 'light',
    `Language` VARCHAR(5) NOT NULL DEFAULT 'pt',
    `UseCamera` TINYINT(1) NOT NULL DEFAULT 0,
    `EnableBonusSystem` TINYINT(1) NOT NULL DEFAULT 0,
    `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `UpdatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`Id`),
    UNIQUE INDEX `UX_user_settings_useruid` (`UserUid` ASC),
    CONSTRAINT `fk_user_settings_users` FOREIGN KEY (`UserUid`) REFERENCES `users` (`Uid`) ON DELETE CASCADE
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `bonus_achievements` (
    `Id` CHAR(36) NOT NULL,
    `Code` VARCHAR(50) NOT NULL,
    `Name` VARCHAR(100) NOT NULL,
    `Description` VARCHAR(255) NULL,
    `PointsRequired` INT NOT NULL,
    `IsActive` TINYINT(1) NOT NULL DEFAULT 1,
    `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`Id`),
    UNIQUE INDEX `UX_bonus_achievements_code` (`Code` ASC)
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `user_bonus_points` (
    `Id` CHAR(36) NOT NULL,
    `UserUid` CHAR(36) NOT NULL,
    `Points` INT NOT NULL,
    `Source` VARCHAR(100) NULL,
    `Description` VARCHAR(255) NULL,
    `CreatedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`Id`),
    INDEX `IX_user_bonus_points_useruid` (`UserUid` ASC),
    CONSTRAINT `fk_user_bonus_points_users` FOREIGN KEY (`UserUid`) REFERENCES `users` (`Uid`) ON DELETE CASCADE
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `user_achievements` (
    `Id` CHAR(36) NOT NULL,
    `UserUid` CHAR(36) NOT NULL,
    `BonusAchievementId` CHAR(36) NOT NULL,
    `PointsAtAchievement` INT NOT NULL,
    `AchievedAt` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`Id`),
    UNIQUE INDEX `UX_user_achievements_user_achievement` (`UserUid` ASC, `BonusAchievementId` ASC),
    CONSTRAINT `fk_user_achievements_users` FOREIGN KEY (`UserUid`) REFERENCES `users` (`Uid`) ON DELETE CASCADE,
    CONSTRAINT `fk_user_achievements_bonus_achievements` FOREIGN KEY (`BonusAchievementId`) REFERENCES `bonus_achievements` (`Id`) ON DELETE RESTRICT
) ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;
